\begin{frame}{The CAST mechanism}
  \centering
  {\Large$\displaystyle
    \textcolor{activationcolor}{\mathbf{h}}' \leftarrow
    \textcolor{activationcolor}{\mathbf{h}}
    + \underbrace{\textcolor{conditioncolor}{f}\big(\mathrm{sim}(
      \textcolor{activationcolor}{\mathbf{h}},\,
      \mathrm{proj}_{\textcolor{conditioncolor}{\mathbf{c}}}
      \textcolor{activationcolor}{\mathbf{h}})\big)}_{%
      \text{\large\color{conditioncolor}switch: 0 or 1}}
    \cdot
    \underbrace{\alpha\cdot\textcolor{accent}{\mathbf{v}}}_{%
      \text{\large\color{accent}behavior}}$\par}
  \vspace{6mm}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.40\textwidth}
      \centering
      {\large$\displaystyle \textcolor{conditioncolor}{f}(s) =
        \begin{cases}
          \textcolor{accent}{1} & \text{if } s > \textcolor{conditioncolor}{\theta}\\
          0 & \text{otherwise}
        \end{cases}$\par}
      \vspace{4mm}
      {\large\color{muted}$s$: cosine similarity between\\
        $\mathbf{h}$ and its projection onto $\mathbf{c}$\par}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \begin{tikzpicture}[font=\large]
        \draw[->,muted!60] (0,0) -- (6.4,0) node[below left,text=muted] {$s$};
        \draw[->,muted!60] (0,0) -- (0,2.4)
          node[left] {$\textcolor{conditioncolor}{f}$};
        \node[left,text=muted] at (0,0) {0};
        \node[left,text=muted] at (0,1.7) {1};
        \draw[dashed,muted!50] (3,0) -- (3,1.7);
        \node[below=1.5mm,text=conditioncolor] at (3,0) {$\theta$};
        \draw[ultra thick,muted] (0,0) -- (3,0);
        \draw[ultra thick,accent] (3,1.7) -- (6,1.7);
        \fill[muted] (3,0) circle (1.4mm);
        \draw[accent,thick,fill=white] (3,1.7) circle (1.4mm);
        \node[above,text=accent] at (4.5,1.75) {add $\alpha\mathbf{v}$};
        \node[above,text=muted] at (1.5,0.05) {unchanged};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \raggedright
  \papersource{Sec.~3.1; cosine similarity for direction: Hsu et al., 2024}
\end{frame}

\begin{frame}{Conclusion: three different problems}
  \centering
  \vspace{5mm}
  \begin{tikzpicture}[

    \node[blk,draw=activationcolor,fill=activationcolor!12] (a) at (0,0)
      {{\large\bfseries 1.\ Information in activations}\\
      {\normalsize\color{muted}Linear probes classify the labels well.}};
    \node[blk,draw=conditioncolor,fill=conditioncolor!10] (b) at (0,-1.65)
      {{\large\bfseries 2.\ Robust concept detection}\\
      {\normalsize\color{muted}Context changes can break the gate.}};
    \node[blk,draw=accent,fill=wash] (c) at (0,-3.3)
      {{\large\bfseries 3.\ Reliable behavioral control}\\
      {\normalsize\color{muted}Steering depends on strength and model.}};
    \draw[flow] (a) -- (b);
    \draw[flow] (b) -- (c);
    \node[tag,text=activationcolor] at (-4.6,0) {Representation};
    \node[tag,text=conditioncolor] at (-4.6,-1.65) {Robustness};
    \node[tag,text=accent] at (-4.6,-3.3) {Control};
  \end{tikzpicture}\par
  \vspace{5mm}
  {\Large\color{accent}A decodable representation is not necessarily\\
    a robust detector or an effective intervention.\par}
  \raggedright
  \oursource{four models, frozen protocol, held-out test prompts}
  \note[item]{About 45 s. Mirrors the opening slide.}
  \note[item]{1: The hidden states contain highly predictive information about
      the dataset labels.}
  \note[item]{2: Detecting that concept reliably when the context changes is a
      separate problem; a fixed direction and threshold can fail in either
      direction.}
  \note[item]{3: Using a vector to change behavior is a further, causal
      problem. It depends on the model, layers and strength.}
  \note[item]{This lesson applies to representation engineering and activation
      steering in general, not just CAST.}
  \note[item]{CAST is an elegant modular design. Our results suggest each
      module should be evaluated on its own: detector, robustness, and
      behavior.}
\end{frame}

  \vspace{2mm}

    \node[font=\Large] at (6.6,0.6) {$\cos(\mathbf{h},\,\mathbf{c})$};
    \node[font=\fontsize{30}{34}\selectfont\bfseries,text=muted]
      at (0,-0.85) {0.619};
    \node[font=\fontsize{30}{34}\selectfont\bfseries,text=conditioncolor]
      at (6.6,-0.85) {0.736};
    \node[font=\large,text=muted] at (0,-1.7) {F1};

  \note[item]{Backup slide; use only if asked.}
  \note[item]{Same condition vectors, features and layer (block 8); each score
      gets its own threshold and direction chosen on validation.}
  \note[item]{Direct cosine beat CAST's similarity in all four float64

  \note[item]{Precision matters: FP16 dot products and norms in the projection

      threshold decisions.}
