\begin{frame}{The CAST mechanism}
  \centering
  {\Large$\displaystyle
    \textcolor{activationcolor}{\mathbf{h}}' \leftarrow
    \textcolor{activationcolor}{\mathbf{h}}
    + \underbrace{\textcolor{conditioncolor}{f}\big(\mathrm{sim}(
      \textcolor{activationcolor}{\mathbf{h}},\,
      \mathrm{proj}_{\textcolor{conditioncolor}{\mathbf{c}}}
      \textcolor{activationcolor}{\mathbf{h}})\big)}_{%
      \text{\large\color{conditioncolor}switch: 0 or 1}}
    \cdot
    \underbrace{\alpha\cdot\textcolor{accent}{\mathbf{v}}}_{%
      \text{\large\color{accent}behavior}}$\par}
  \vspace{6mm}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.40\textwidth}
      \centering
      {\large$\displaystyle \textcolor{conditioncolor}{f}(s) =
        \begin{cases}
          \textcolor{accent}{1} & \text{if } s > \textcolor{conditioncolor}{\theta}\\
          0 & \text{otherwise}
        \end{cases}$\par}
      \vspace{4mm}
      {\large\color{muted}$s$: cosine similarity between\\
        $\mathbf{h}$ and its projection onto $\mathbf{c}$\par}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \begin{tikzpicture}[font=\large]
        \draw[->,muted!60] (0,0) -- (6.4,0) node[below left,text=muted] {$s$};
        \draw[->,muted!60] (0,0) -- (0,2.4)
          node[left] {$\textcolor{conditioncolor}{f}$};
        \node[left,text=muted] at (0,0) {0};
        \node[left,text=muted] at (0,1.7) {1};
        \draw[dashed,muted!50] (3,0) -- (3,1.7);
        \node[below=1.5mm,text=conditioncolor] at (3,0) {$\theta$};
        \draw[ultra thick,muted] (0,0) -- (3,0);
        \draw[ultra thick,accent] (3,1.7) -- (6,1.7);
        \fill[muted] (3,0) circle (1.4mm);
        \draw[accent,thick,fill=white] (3,1.7) circle (1.4mm);
        \node[above,text=accent] at (4.5,1.75) {add $\alpha\mathbf{v}$};
        \node[above,text=muted] at (1.5,0.05) {unchanged};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \raggedright
  \papersource{Sec.~3.1; cosine similarity for direction: Hsu et al., 2024}
\end{frame}
