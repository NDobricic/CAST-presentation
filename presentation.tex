\begin{frame}{Control when steering is applied}
  Activation steering adds a behavior vector to a model's hidden state.
  CAST adds a condition that decides when to apply it.

  \vspace{7mm}
  \begin{center}
  \begin{tikzpicture}[
    box/.style={draw=accent,fill=wash,rounded corners=3pt,
      minimum height=13mm,text width=31mm,align=center,font=\small},
    flow/.style={-{Latex[length=2mm]},thick,draw=accent},
    node distance=10mm]
    \node[box] (state) {Prompt\\activations};
    \node[box,right=of state] (gate) {Condition check\\$g\in\{0,1\}$};
    \node[box,right=of gate] (steer) {Apply behavior\\if $g=1$};
    \draw[flow] (state) -- (gate);
    \draw[flow] (gate) -- (steer);
  \end{tikzpicture}
  \end{center}
  \vspace{3mm}
  \[
    h' = h + g\,\alpha\,v
  \]
  {\small $h$: hidden state\quad $v$: behavior vector\quad
   $\alpha$: steering strength}
  \papersource{Section 3.1; schematic notation}
\end{frame}
